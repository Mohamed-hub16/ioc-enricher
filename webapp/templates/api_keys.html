Exemples d'utilisation 
# Lister les IOCs (page 1, 50 par page)
curl -H "X-API-Key: <votre-token>" \
     "{{ request.host_url }}api/v1/iocs?type=ip&min_score=50"

# Détail d'un IOC
curl -H "X-API-Key: <votre-token>" \
     "{{ request.host_url }}api/v1/ioc/8.8.8.8"

# Enrichir un IOC
curl -X POST \
     -H "X-API-Key: <votre-token>" \
     -H "Content-Type: application/json" \
     -d '{"ioc": "8.8.8.8"}' \
     "{{ request.host_url }}api/v1/enrich"

# Supprimer un IOC (admin uniquement)
curl -X DELETE \
     -H "X-API-Key: <votre-token>" \
     "{{ request.host_url }}api/v1/ioc/8.8.8.8"
